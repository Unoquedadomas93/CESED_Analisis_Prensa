CREATE OR REPLACE TABLE
  `cesed-gdelt-sentencia-1994.sentencia_1994.colombia_events_1993_1995` AS
SELECT
  SQLDATE,
  Year,
  MonthYear,
  Actor1Name,
  Actor1CountryCode,
  Actor2Name,
  Actor2CountryCode,
  EventCode,
  EventRootCode,
  QuadClass,
  GoldsteinScale,
  NumMentions,
  NumSources,
  NumArticles,
  AvgTone,
  ActionGeo_FullName,
  ActionGeo_CountryCode
FROM `gdelt-bq.full.events`
WHERE Year BETWEEN 1993 AND 1995
  AND ActionGeo_CountryCode = 'CO';
